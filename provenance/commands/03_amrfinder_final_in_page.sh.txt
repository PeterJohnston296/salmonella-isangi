output_dir="/data/fast/salmonella/pjohnston/isangi/amrfinder2_out"
run_amrfinder() {
    local fasta_file="$1"
    local base_name=$(basename "$fasta_file" .fasta)
    amrfinder -n "$fasta_file" --organism Salmonella -o "$output_dir/${base_name}_amrfinder.tsv"
}
export -f run_amrfinder
export output_dir
find . -name "*.fasta" | parallel run_amrfinder
