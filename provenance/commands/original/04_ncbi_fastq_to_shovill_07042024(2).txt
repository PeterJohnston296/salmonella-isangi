#!/bin/bash

# Loop through only the "_1.fastq.gz" files to ensure each pair is processed once
 # Extract the base SAMPLE name by removing the "_1.fastq.gz" part
 # Define forward and reverse read files
 # Print the file names (optional, for verification)
 # Create an output directory specifically for this sample
 # Specify the amount of RAM and CPUs based on your system's resources

#!/bin/bash

# Define the output directory for Shovill assemblies
output_dir="shovill_assemblies"

# Create the output directory if it doesn't already exist
mkdir -p "$output_dir"

# Loop through all "_1.fastq.gz" files in the current directory
for fwd_reads in *_1.fastq.gz; do
    # Remove the "_1.fastq.gz" suffix to get the base sample name
    base_name="${fwd_reads%_1.fastq.gz}"
    
    # Define the forward and reverse read files
    fwd_file="${base_name}_1.fastq.gz"
    rev_file="${base_name}_2.fastq.gz"
    
    # Specify the output directory for the current sample
    sample_out_dir="${output_dir}/${base_name}"
    mkdir -p "$sample_out_dir"
    
    # Print processing message
    echo "Processing sample: $base_name"
    echo "Forward reads: $fwd_file"
    echo "Reverse reads: $rev_file"
    echo "Output directory: $sample_out_dir"
    
    # Run Shovill to assemble the genome
    shovill --R1 "$fwd_file" --R2 "$rev_file" \
            --outdir "$sample_out_dir" \
            --cpus 4 --ram 16  # Adjust these values based on your machine's resources
    
    # Check if Shovill completed successfully
    if [ $? -eq 0 ]; then
        echo "Assembly completed successfully for sample: $base_name"
    else
        echo "Error during assembly for sample: $base_name"
    fi
done

echo "All samples have been processed."

#!/bin/bash



# Set the number of parallel jobs
num_jobs=8
# Define the base output directory for all Shovill assemblies
shovill_out_ncbi="/data/fast/salmonella/pjohnston/isangi/NCBI/shovill_out_ncbi"
# Ensure the output directory exists
mkdir -p "$shovill_out_ncbi"
# File containing list of sample identifiers
sample_list="unique_prefixes_2.txt"
# Run Shovill for each sample identifier in parallel and rename output files
cat "$sample_list" | parallel -j $num_jobs \
    "shovill --R1 {}_1.fastq.gz --R2 {}_2.fastq.gz \
            --outdir '$shovill_out_ncbi/{}' \
            --cpus 8 --ram 16; \
     for file in '$shovill_out_ncbi'/{}/*; do \
         mv \"\$file\" \"'$shovill_out_ncbi'/{}\"/\"$(basename {})_\$(basename \"\$file\")\"; \
     done"
echo "All samples have been processed"

# this also worked, but also did not append unique file names

# To change names of files within each subdirectory to take the prefix of the subdirectory (e.g. the SRR / ERR number)


# Define the parent directory containing the subdirectories.
parent_dir="/data/fast/salmonella/pjohnston/isangi/NCBI/shovill_out_ncbi"

# Change into the parent directory
cd "$parent_dir" || { echo "Could not enter directory: $parent_dir"; exit 1; }

# Find all subdirectories in the parent directory
find . -type d -mindepth 1 -maxdepth 1 | while read -r dir; do
    # Get the base name of the directory without leading './'
    dir_name=$(basename "$dir")

    # Loop through each file in the current subdirectory
    for file in "$dir"/*; do
        # Check if it is a file, not a directory
        if [ -f "$file" ]; then
            # Get the base name of the file
            file_name=$(basename "$file")
            # Rename the file to prefix it with the directory name
            mv "$file" "${dir}/${dir_name}_${file_name}"
        fi
    done
done

echo "Renaming complete."
