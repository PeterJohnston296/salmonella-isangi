#!/bin/bash

input_file="/data/fast/salmonella/pjohnston/isangi/NCBI/NCBI_fastq/unique_prefixes_2.txt"
output_file="/data/fast/salmonella/pjohnston/isangi/NCBI/NCBI_fastq/output.txt"


for identifier in $(cat $input_file); do

  result=$(esearch -db sra -query "$identifier" | efetch -format docsum | grep -e SRS -e ERS)

  echo "$identifier $result" >> "$output_file"

done

# This gives me a txt file "output.txt". This contains 274 unique SRS / ERS numbers, but in this format:

<Sample acc="SRS4305031" name=""/>

input_file="/data/fast/salmonella/pjohnston/isangi/NCBI/NCBI_fastq/output.txt"
output_file="/data/fast/salmonella/pjohnston/isangi/NCBI/NCBI_fastq/extracted_srs_ers"

# Use grep to find lines containing the patterns, then sed to extract the desired substring
grep -oE 'SRS[0-9]+|ERS[0-9]+' "$input_file" > "$output_file"

echo "Operation completed. Extracted strings have been written to $output_file."
